# ---------------------------------------------------------------------------
# Shared cube-appearance GLSL, consumed by BOTH backends from one source.
#
# The browser's .vert/.frag #include these files directly (qsb honours
# GL_GOOGLE_include_directive, resolving against the including file's directory). The
# desktop composes its programs from string_views at runtime, so the same files are
# embedded here into a generated header. Without this the key light and the edge band
# were spelled once per shader and nothing made the copies agree.
#
# CONFIGURE_DEPENDS matters: the .qsb staleness check below runs at configure time, so a
# changed helper has to re-trigger configure for the check to see it.
# ---------------------------------------------------------------------------
set(_pj_cube_glsl_dir "${CMAKE_CURRENT_SOURCE_DIR}/shaders/cube")
set(_pj_cube_glsl_files cube_lighting cube_edge cube_fan cube_face_normal cube_sprite)
set(_pj_cube_glsl_header "${CMAKE_CURRENT_BINARY_DIR}/generated/pj_scene3d_widgets/cube_shader_sources.h")

set(_pj_cube_glsl_content "// GENERATED by pj_scene3D/widgets/CMakeLists.txt — do not edit.\n")
string(APPEND _pj_cube_glsl_content
  "// Source of truth: pj_scene3D/widgets/shaders/cube/*.glslinc, which the browser\n"
  "// shaders #include directly. Edit those, not this.\n"
  "#pragma once\n\n#include <string_view>\n\nnamespace pj::scene3d {\n\n")
foreach(_pj_cube_glsl IN LISTS _pj_cube_glsl_files)
  set(_pj_cube_glsl_path "${_pj_cube_glsl_dir}/${_pj_cube_glsl}.glslinc")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_pj_cube_glsl_path}")
  file(READ "${_pj_cube_glsl_path}" _pj_cube_glsl_text)
  # snake_case file -> CamelCase symbol: cube_face_normal -> kCubeFaceNormalGlsl.
  string(REPLACE "cube_" "" _pj_cube_glsl_stem "${_pj_cube_glsl}")
  string(REPLACE "_" ";" _pj_cube_glsl_words "${_pj_cube_glsl_stem}")
  set(_pj_cube_glsl_symbol "")
  foreach(_pj_cube_glsl_word IN LISTS _pj_cube_glsl_words)
    string(SUBSTRING "${_pj_cube_glsl_word}" 0 1 _pj_cube_glsl_first)
    string(SUBSTRING "${_pj_cube_glsl_word}" 1 -1 _pj_cube_glsl_rest)
    string(TOUPPER "${_pj_cube_glsl_first}" _pj_cube_glsl_first)
    string(APPEND _pj_cube_glsl_symbol "${_pj_cube_glsl_first}${_pj_cube_glsl_rest}")
  endforeach()
  string(APPEND _pj_cube_glsl_content
    "inline constexpr std::string_view kCube${_pj_cube_glsl_symbol}Glsl = R\"PJGLSL(\n"
    "${_pj_cube_glsl_text}"
    ")PJGLSL\";\n\n")
endforeach()
string(APPEND _pj_cube_glsl_content "}  // namespace pj::scene3d\n")
file(GENERATE OUTPUT "${_pj_cube_glsl_header}" CONTENT "${_pj_cube_glsl_content}")
