# ---------------------------------------------------------------------------
# Shared cube-appearance GLSL, one source for every shader that draws cubes.
#
# The .vert/.frag sources #include these files directly (qsb honours
# GL_GOOGLE_include_directive, resolving against the including file's directory). The
# same files are embedded here into a generated header, so the C++ mirrors of their
# constants (cube_mesh.h) are checked against the GLSL text by the tests.
#
# CONFIGURE_DEPENDS matters: the .qsb staleness check below runs at configure time, so a
# changed helper has to re-trigger configure for the check to see it.
# ---------------------------------------------------------------------------
set(_pj_cube_glsl_dir "${CMAKE_CURRENT_SOURCE_DIR}/shaders/cube")
set(_pj_cube_glsl_files cube_lighting cube_edge cube_fan cube_face_normal cube_sprite cube_point_color)
set(_pj_cube_glsl_header "${CMAKE_CURRENT_BINARY_DIR}/generated/pj_scene3d_widgets/cube_shader_sources.h")

set(_pj_cube_glsl_content "// GENERATED by pj_scene3D/widgets/CMakeLists.txt — do not edit.\n")
string(APPEND _pj_cube_glsl_content
  "// Source of truth: pj_scene3D/widgets/shaders/cube/*.glslinc, which the\n"
  "// shaders #include directly. Edit those, not this.\n"
  "#pragma once\n\n#include <string_view>\n\nnamespace pj::scene3d {\n\n")
foreach(_pj_cube_glsl IN LISTS _pj_cube_glsl_files)
  set(_pj_cube_glsl_path "${_pj_cube_glsl_dir}/${_pj_cube_glsl}.glslinc")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_pj_cube_glsl_path}")
  file(READ "${_pj_cube_glsl_path}" _pj_cube_glsl_text)
  # snake_case file -> CamelCase symbol: cube_face_normal -> kCubeFaceNormalGlsl.
  string(REPLACE "cube_" "" _pj_cube_glsl_stem "${_pj_cube_glsl}")
  string(REPLACE "_" ";" _pj_cube_glsl_words "${_pj_cube_glsl_stem}")
  set(_pj_cube_glsl_symbol "")
  foreach(_pj_cube_glsl_word IN LISTS _pj_cube_glsl_words)
    string(SUBSTRING "${_pj_cube_glsl_word}" 0 1 _pj_cube_glsl_first)
    string(SUBSTRING "${_pj_cube_glsl_word}" 1 -1 _pj_cube_glsl_rest)
    string(TOUPPER "${_pj_cube_glsl_first}" _pj_cube_glsl_first)
    string(APPEND _pj_cube_glsl_symbol "${_pj_cube_glsl_first}${_pj_cube_glsl_rest}")
  endforeach()
  string(APPEND _pj_cube_glsl_content
    "inline constexpr std::string_view kCube${_pj_cube_glsl_symbol}Glsl = R\"PJGLSL(\n"
    "${_pj_cube_glsl_text}"
    ")PJGLSL\";\n\n")
endforeach()
string(APPEND _pj_cube_glsl_content "}  // namespace pj::scene3d\n")
file(GENERATE OUTPUT "${_pj_cube_glsl_header}" CONTENT "${_pj_cube_glsl_content}")

# carries the host qsb (Qt ShaderTools); desktop kits don't install it.
if(EMSCRIPTEN)
  include(${CMAKE_SOURCE_DIR}/cmake/PjQsb.cmake)
  pj_find_host_qsb(_pj_scene3d_qsb "WASM Scene3D")

  # bake.sh is the single source of the qsb targets; re-run it and require the
  # committed packs to match byte for byte.
  set(_pj_scene3d_shader_dir "${CMAKE_CURRENT_SOURCE_DIR}/rhi/shaders")
  set(_pj_scene3d_shader_verify_dir "${CMAKE_CURRENT_BINARY_DIR}/verified_shaders")
  file(REMOVE_RECURSE "${_pj_scene3d_shader_verify_dir}")
  file(MAKE_DIRECTORY "${_pj_scene3d_shader_verify_dir}")
  execute_process(
    COMMAND bash "${_pj_scene3d_shader_dir}/bake.sh" "${_pj_scene3d_qsb}" "${_pj_scene3d_shader_verify_dir}"
    RESULT_VARIABLE _pj_scene3d_qsb_result
    ERROR_VARIABLE _pj_scene3d_qsb_error)
  if(NOT _pj_scene3d_qsb_result EQUAL 0)
    message(FATAL_ERROR "Failed to bake the Scene3D shaders: ${_pj_scene3d_qsb_error}")
  endif()
  file(GLOB _pj_scene3d_rebuilt_packs RELATIVE "${_pj_scene3d_shader_verify_dir}"
    "${_pj_scene3d_shader_verify_dir}/*.qsb")
  file(GLOB _pj_scene3d_committed_packs RELATIVE "${_pj_scene3d_shader_dir}" "${_pj_scene3d_shader_dir}/*.qsb")
  if(NOT _pj_scene3d_committed_packs STREQUAL _pj_scene3d_rebuilt_packs)
    message(FATAL_ERROR "Scene3D shader packs out of sync with rhi/shaders/bake.sh: committed "
      "[${_pj_scene3d_committed_packs}], baked [${_pj_scene3d_rebuilt_packs}]")
  endif()
  foreach(_pj_scene3d_pack IN LISTS _pj_scene3d_rebuilt_packs)
    file(SHA256 "${_pj_scene3d_shader_dir}/${_pj_scene3d_pack}" _pj_scene3d_committed_hash)
    file(SHA256 "${_pj_scene3d_shader_verify_dir}/${_pj_scene3d_pack}" _pj_scene3d_rebuilt_hash)
    if(NOT _pj_scene3d_committed_hash STREQUAL _pj_scene3d_rebuilt_hash)
      message(FATAL_ERROR
        "Stale Scene3D shader pack ${_pj_scene3d_pack}; re-bake with rhi/shaders/bake.sh")
    endif()
  endforeach()
  file(GLOB _pj_scene3d_shader_sources "${_pj_scene3d_shader_dir}/*.vert" "${_pj_scene3d_shader_dir}/*.frag"
    "${_pj_scene3d_shader_dir}/*.glslinc")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${_pj_scene3d_shader_dir}/bake.sh" ${_pj_scene3d_shader_sources})
endif()
